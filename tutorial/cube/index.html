    A cube from the disc

    The second program draws a cube and turns it with the directional buttons. The cube is a model, and its texture is a picture, and both are files on the disc that the program reads at startup, which is how a program gets anything bigger than its own code into memory.

    It starts from the module setting up made; a new directory with the same go.mod does as well.

    The files

    A model for the PlayStation is a TMD, the format Sony's tools wrote: a list of vertices and normals, and a list of polygons that index them, each with its color or its texture coordinates. A texture is a TIM, a picture in one of the GPU's formats with the place in video memory it is to be loaded to, and its palette if it has one. Save these two beside the program, in a data directory:

    • CUBE.TMD, a cube 128 units along each edge: eight vertices, six normals, and six textured polygons, one for each face.
    • CUBE.TIM, a 64 by 64 texture in 16 colors, with its palette.

    The texture comes from Sony's own Net Yaroze tutorials, by way of the flying example, which tiles the plane it flies over with it. The cube is written by a short script, which is the whole of the TMD format for a model this simple, and the program, its files and the script are in the repository at docs/tutorial/cube.

    The program

    //go:build yaroze

// Cube: a textured cube from the disc, turned with the directional buttons.
package main

import (
	"unsafe"

	"go.chrisrx.dev/psx/disc"
	"go.chrisrx.dev/psx/yaroze"
)

const (
	screenW, screenH = 320, 240

	// The ordering table holds 1<<otLength depths; one object needs few.
	otLength  = 4
	packetMax = 4096

	// How far a held button turns the cube each frame, where 4096 is a
	// whole turn.
	step = 24
)

// The files on the disc, and the memory they are read into. The drive
// delivers whole 2048-byte sectors, so each buffer is a whole number of them,
// enough for its file.
var (
	model   = disc.Register("DATA/CUBE.TMD", "data/CUBE.TMD")
	texture = disc.Register("DATA/CUBE.TIM", "data/CUBE.TIM")

	modelData   [2048 / 4]uint32
	textureData [4096 / 4]uint32
)

var (
	ot         [2]yaroze.GsOT
	otTags     [2][1 << otLength]yaroze.GsOT_TAG
	packetArea [2][packetMax]yaroze.PACKET

	view  yaroze.GsRVIEW2
	light yaroze.GsF_LIGHT
	cube  yaroze.GsDOBJ2
	coord yaroze.GsCOORDINATE2
	image yaroze.GsIMAGE

	// rotation is which way the cube faces, built up turn by turn; turned
	// and ls are the frame's working matrices.
	rotation, turned, ls yaroze.MATRIX

	padBuf0, padBuf1 *uint8
	pathBuf          [64]byte
	textBuf          [64]byte
)

// The directional buttons, as bits of what padRead returns.
const (
	padUp    = 1 << 12
	padRight = 1 << 13
	padDown  = 1 << 14
	padLeft  = 1 << 15
)

func main() {
	yaroze.SetVideoMode(yaroze.MODE_NTSC)
	yaroze.GsInitGraph(screenW, screenH, yaroze.GsNONINTER|yaroze.GsOFSGPU, 1, 0)
	yaroze.GsDefDispBuff(0, 0, 0, screenH)
	yaroze.GsInit3D()
	for i := range ot {
		ot[i].Length = otLength
		ot[i].Org = &otTags[i][0]
	}
	yaroze.FntLoad(960, 256)
	// With GsOFSGPU the origin is the middle of the screen, for 3D's sake,
	// so the text window is placed from there.
	yaroze.FntOpen(-screenW/2+16, -screenH/2+16, 256, 64, 0, 512)
	yaroze.GetPadBuf(&padBuf0, &padBuf1)

	if !load(&model, &modelData[0]) || !load(&texture, &textureData[0]) {
		yaroze.FntPrint(yaroze.CString(textBuf[:], "could not read the disc"))
		for {
			yaroze.FntFlush(-1)
			yaroze.VSync(0)
		}
	}

	// The texture goes into video memory, at the place the file names and
	// the model's polygons point to, with its palette.
	yaroze.GsGetTimInfo(&textureData[1], &image)
	rect := yaroze.RECT{X: image.Px, Y: image.Py, W: int16(image.Pw), H: int16(image.Ph)}
	yaroze.LoadImage(&rect, image.Pixel)
	if image.Pmode&8 != 0 {
		rect = yaroze.RECT{X: image.Cx, Y: image.Cy, W: int16(image.Cw), H: int16(image.Ch)}
		yaroze.LoadImage(&rect, image.Clut)
	}
	yaroze.DrawSync(0)

	// The model's offsets become addresses now that it has one, and the
	// object is linked to the first model in the file, in its own
	// coordinate system.
	yaroze.GsMapModelingData(&modelData[1])
	yaroze.GsLinkObject4(uint32(uintptr(unsafe.Pointer(&modelData[3]))), &cube, 0)
	yaroze.GsInitCoordinate2(nil, &coord)
	cube.Coord2 = &coord

	// The camera: back from the origin and a little above it, looking at it.
	view = yaroze.GsRVIEW2{Vpx: 0, Vpy: -120, Vpz: -320}
	yaroze.GsSetProjection(240)
	yaroze.GsSetRefView2(&view)

	// One light from over the camera's shoulder, and some ambient light,
	// so that the faces turned away from it are not black.
	light = yaroze.GsF_LIGHT{Vx: 1, Vy: 1, Vz: 1, R: 0xc0, G: 0xc0, B: 0xc0}
	yaroze.GsSetFlatLight(0, &light)
	yaroze.GsSetAmbient(yaroze.ONE/4, yaroze.ONE/4, yaroze.ONE/4)
	yaroze.GsSetLightMode(0)

	// Facing straight ahead, with a little of the top and a side showing.
	rotation = yaroze.MATRIX{M: [3][3]int16{{yaroze.ONE, 0, 0}, {0, yaroze.ONE, 0}, {0, 0, yaroze.ONE}}}
	turn(&yaroze.SVECTOR{Vx: -step * 12, Vy: step * 12})

	hint := yaroze.CString(textBuf[:], "the directions turn the cube")
	for {
		pad := padRead()
		if pad&padUp != 0 {
			turn(&yaroze.SVECTOR{Vx: -step})
		}
		if pad&padDown != 0 {
			turn(&yaroze.SVECTOR{Vx: step})
		}
		if pad&padLeft != 0 {
			turn(&yaroze.SVECTOR{Vy: -step})
		}
		if pad&padRight != 0 {
			turn(&yaroze.SVECTOR{Vy: step})
		}
		coord.Coord = rotation
		coord.Flg = 0

		active := yaroze.GsGetActiveBuff()
		yaroze.GsSetWorkBase(&packetArea[active][0])
		yaroze.GsClearOt(0, 0, &ot[active])

		yaroze.GsGetLs(&coord, &ls)
		yaroze.GsSetLightMatrix(&ls)
		yaroze.GsSetLsMatrix(&ls)
		yaroze.GsSortObject4(&cube, &ot[active], 14-otLength, scratch)

		yaroze.DrawSync(0)
		yaroze.VSync(0)
		yaroze.GsSwapDispBuff()
		yaroze.GsSortClear(60, 120, 120, &ot[active])
		yaroze.GsDrawOt(&ot[active])
		yaroze.FntPrint(hint)
		yaroze.FntFlush(-1)
	}
}

// turn rotates the cube by the angles given about the world's axes, which
// are the screen's: the new turn is applied after everything so far, so up
// always tips the top towards the camera and left always turns the left
// side away, however the cube already sits.
func turn(by *yaroze.SVECTOR) {
	var m yaroze.MATRIX
	yaroze.RotMatrix(by, &m)
	yaroze.MulMatrix0(&m, &rotation, &turned)
	rotation = turned
}

// scratch is the CPU's scratchpad, 1KB of fast memory that GsSortObject4
// uses as its work space.
var scratch = (*uint32)(unsafe.Pointer(uintptr(0x1f800000)))

// load reads a file off the disc into memory and waits for it, giving up
// after a few seconds so that a missing disc shows a message rather than
// hanging.
func load(f *disc.File, into *uint32) bool {
	if !f.Read(into, pathBuf[:]) {
		return false
	}
	for range 300 {
		if yaroze.CdReadSync(1, nil) <= 0 {
			return true
		}
		yaroze.VSync(0)
	}
	return false
}

// padRead returns the buttons held on the first controller, one bit each,
// from the buffers the library refreshes every frame.
func padRead() uint32 {
	if padBuf0 == nil {
		return 0
	}
	at := func(n uintptr) uint32 { return uint32(*(*uint8)(unsafe.Add(unsafe.Pointer(padBuf0), n))) }
	return ^(at(3) | at(2)<<8) & 0xffff
}

    Getting the files onto the disc

    Two lines near the top do more than they look like they do:

    model   = disc.Register("DATA/CUBE.TMD", "data/CUBE.TMD")
texture = disc.Register("DATA/CUBE.TIM", "data/CUBE.TIM")

    Each names a path on the disc and the file on the host that goes there. TinyGo works the call out when it compiles the program, so each registration ends up as a small record of constant data in the executable. When mkpsxiso builds the disc it finds the records in the executable it is told to boot, places each file at its path, and once it knows where on the disc each one landed writes the file's first sector and its length back into the record. The program is then built with the knowledge of where its files are, and Read reads a file from that sector directly, without searching the disc's directory for it by name. A file that is missing or misnamed fails the build rather than the program.

    load wraps Read with the wait every disc read needs. The drive works in the background: Read starts it and returns, and CdReadSync reports whether it has finished, so the program polls it once a frame. Giving up after a few seconds turns a missing disc into a message instead of a hang.

    The two buffers the files are read into are whole numbers of 2048-byte sectors, because that is what the drive delivers, whatever the file's length.

    Drawing in three dimensions

    The library draws models through objects. A GsDOBJ2 ties a model in a TMD to a coordinate system, a GsCOORDINATE2, which holds a matrix saying where the object is and which way it faces. The program links its one object to the one model in the file, after GsMapModelingData has turned the file's offsets into addresses, since the file could have been loaded anywhere.

    The cube's orientation is a matrix the program keeps, rotation, and turn changes it: RotMatrix makes a matrix for a small turn, with 4096 standing for a whole turn as it does everywhere in the library, and MulMatrix0 multiplies it onto the front of what is there. Multiplying on the front is what makes the controls behave: the new turn is applied after every turn so far, so it is always about the screen's axes, and up always tips the top towards the camera however the cube already sits. Multiplying on the back instead would turn the cube about its own axes, which feels right until it has been tipped over. Each frame the matrix is copied into the coordinate system and Flg set to zero, which tells the library the coordinates changed, so that it works out the object's screen matrix again. GsGetLs fetches that, GsSetLsMatrix hands it to the GTE, the geometry coprocessor that does the arithmetic in fixed point with twelve fractional bits, and GsSortObject4 turns the model into drawing commands and sorts them into the ordering table by depth.

    Only the faces turned towards the camera are drawn. The GPU decides that by which way round a polygon's corners come out on the screen, so a TMD's polygons have to be written with their corners in a consistent order as seen from outside, which the script that makes the cube is careful about.

    The camera is a GsRVIEW2: a viewpoint, a point it looks at, and a twist about that line. Here it sits back from the origin and a little above it, looking at the origin, and GsSetProjection sets how strong the perspective is, as the distance from the eye to the screen in the same units. With GsOFSGPU the screen's origin is its middle, which three dimensions want, so the text window's position is counted from there.

    The light is the simplest the library offers, one flat light shining along a direction, with some ambient light so that the faces turned away from it are not black. Each face has a normal, so each shades by its angle to the light as the cube turns, which is what makes it read as a solid.

    Building and running it

    The same steps as before, with the two files joining the disc through --src, the directory their registered paths are relative to:

    tinygo build -target=playstation-yaroze -o prog.elf .
go tool mips1fix prog.elf
go tool eco2exe prog.elf MAIN.EXE
go tool mkpsxiso -v -o cube.bin --boot MAIN.EXE --file MAIN.EXE=MAIN.EXE --src .
go tool psxemu --bios openbios.bin --disc cube.cue --png cube.png

    -v lists what was placed where, with the two registered files among it, and a run in mednafen or the console shows the cube, which the directions turn.
